Ventes et paiements

Quand, combien et comment la boutique vend : chiffre d’affaires et commandes, produits, livraison, codes promo, puis paiements et versements.

Ce que ça mesure
Le chiffre d’affaires et le nombre de commandes dans le temps (mois, jour de semaine, heure), les produits vendus, les modes de livraison, les codes promo, les moyens de paiement et leurs frais.
Sur quel périmètre
1 438 commandes non annulées du 27 juin 2025 au 20 septembre 2026 (7 annulées et la ligne « pourboire » exclues). Montants facturés : la TVA n’est facturée que depuis le 28 mai 2026.
Ce que ça permet de conclure
Repérer les pics, les moments de la semaine et de la journée où les clients commandent, les produits et les modes de livraison dominants, le coût réel des paiements.
Ce que ça ne permet pas de conclure
Dire pourquoi un pic a lieu (voir la page Explications), ni mesurer la rentabilité complète : seule la marge sur les bougies (50 %) est connue. Septembre 2026 est partiel (données jusqu’au 20).

Dans les blocs ci-dessous, une pastille dit à chaque ligne de quoi il s’agit : Fait (chiffre calculé dans les fichiers), Interprétation possible (explication non démontrée), Pour trancher (ce qu’il faudrait savoir) et Limite (ce qui peut fausser la lecture).

Les agrégats de base de cette page (chiffre d’affaires par mois, commandes par jour et par heure, codes promo, livraison, paiements) ont été produits par un premier jeu de scripts qui n’a pas été conservé. Leurs chiffres clés sont recalculés et contrôlés par le script ci-dessous (voir aussi la page Méthode).

Sert de contrôle : si une ligne affiche « ÉCART », le chiffre du site doit être revu."""
import csv, os, re
from collections import Counter, OrderedDict
from datetime import datetime

# Bloc 1 : pages « Premières analyses » = export de septembre (au 20/09/2026). Bloc 2 : pages mises à jour = livraison d'octobre (au 22/10/2026).
D = '../DONNEES/archive_septembre/orders.csv'; DL = '../DONNEES/archive_septembre/'; DP = '../DONNEES/'
rows = list(csv.DictReader(open(D, encoding='utf-8')))
num = lambda x: float(x.replace(',', '.')) if x not in ('', None) else 0.0
cmd = OrderedDict()
for r in rows:
    if r['order_key'] not in cmd: cmd[r['order_key']] = r          # une ligne d'en-tête de commande = la première ligne de chaque clé
tous = list(cmd.values())
valides = [r for r in tous if not r['Cancelled at']]

resultats = []
def controle(nom, calcule, affiche, tol=0.0):
    ok = abs(calcule - affiche) <= tol
    resultats.append((nom, calcule, affiche, 'OK' if ok else 'ÉCART'))

controle('Commandes non annulées', len(valides), 1438)
controle('Commandes annulées (exclues)', len(tous) - len(valides), 7)
ca = sum(num(r['Total']) for r in valides)
controle('Chiffre d\'affaires facturé (€)', round(ca), 90721, 1)
controle('Panier moyen (€)', round(ca / len(valides), 2), 63.09, 0.01)
avec_code = sum(1 for r in valides if r['Discount Code'])
controle('Commandes avec un code promo (tous codes)', avec_code, 516)
controle('Part des commandes avec code (%)', round(100 * avec_code / len(valides), 1), 35.9, 0.05)
fr = sum(1 for r in valides if (r['Shipping Country'] or r['Billing Country']) == 'FR')
controle('Commandes livrées en France', fr, 1350)
jours = Counter(datetime.strptime(r['Created at'][:10], '%Y-%m-%d').weekday() for r in valides)
controle('Part du dimanche (%)', round(100 * jours[6] / len(valides), 1), 21.3, 0.05)
heures = Counter(int(r['Created at'][11:13]) for r in valides)
h, n = heures.most_common(1)[0]
controle('Heure la plus chargée (10 h) : part (%)', round(100 * n / len(valides), 1) if h == 10 else -1, 8.6, 0.05)
dates = sorted(r['Created at'][:10] for r in valides)
controle('Première commande = 2025-06-27', 1 if dates[0] == '2025-06-27' else 0, 1)
controle('Dernière commande = 2026-09-20', 1 if dates[-1] == '2026-09-20' else 0, 1)
sept25 = sum(1 for d in dates if '2025-09-01' <= d <= '2025-09-20'); sept26 = sum(1 for d in dates if '2026-09-01' <= d <= '2026-09-20')
controle('Commandes du 1er au 20 sept. 2025', sept25, 114); controle('Commandes du 1er au 20 sept. 2026', sept26, 163)
ca_s = lambda a, b: sum(num(r['Total']) for r in valides if a <= r['Created at'][:10] <= b)
controle('CA facturé 1-20 sept. 2025 (€)', round(ca_s('2025-09-01', '2025-09-20')), 7491, 1); controle('CA facturé 1-20 sept. 2026 (€)', round(ca_s('2026-09-01', '2026-09-20')), 10820, 1)
sans_cle = sum(1 for r in valides if not r['customer_key'])
controle('Commandes sans identifiant client', sans_cle, 25)
tva = sum(num(r['Taxes']) for r in valides)
controle('TVA collectée (€)', round(tva), 3685, 1)
ht = ca - tva
controle('CA hors TVA (€)', round(ht), 87036, 1)
premiere_tva = min(r['Created at'][:10] for r in valides if num(r['Taxes']) > 0)
controle('Première commande avec TVA = 2026-05-28', 1 if premiere_tva == '2026-05-28' else 0, 1)
# fichier clients

controle('Total dépensé des fiches clients (€)', round(sum(num(r['Total Spent']) for r in cl)), 89718, 1)

behavior.py · Segments de clientèle (récence, fréquence, montant), cohortes de réachat, produits achetés ensemble.

import csv, json
from datetime import datetime, date
from collections import defaultdict

def parse_dt(s):
    # "2026-09-20 20:42:31 +0200"
    return datetime.strptime(s[:19], "%Y-%m-%d %H:%M:%S")

orders = []
with open('../DONNEES/orders.csv', encoding='utf-8') as f:
    r = csv.DictReader(f)
    cur = None
    for row in r:
        if row['Total']:
            cur = {'r': row, 'items': []}
            orders.append(cur)
        if cur: cur['items'].append(row)
orders = [o for o in orders if not o['r'].get('Cancelled at')]
for o in orders:
    o['dt'] = parse_dt(o['r']['Created at'])
    o['tot'] = float(o['r']['Total'])
    o['key'] = o['r']['customer_key'].strip()

REF = datetime(2026,9,21)  # lendemain de la dernière commande (20/09/2026)

cust = defaultdict(list)
for o in orders:
    if o['key']:
        cust[o['key']].append(o)
for v in cust.values():
    v.sort(key=lambda o:o['dt'])

# ---- RFM pragmatique (4 segments + VIP transversal)
rows = []
for k, os_ in cust.items():
    freq = len(os_)
    monetary = sum(o['tot'] for o in os_)
    recency = (REF - os_[-1]['dt']).days
    first = os_[0]['dt']
    rows.append({'key':k,'freq':freq,'monetary':monetary,'recency':recency,'first':first})

def seg(r):
    if r['freq'] == 1 and r['recency'] > 180: return "Perdue après 1 achat"
    if r['freq'] == 1: return "Nouvelle (1 achat récent)"
    if r['freq'] >= 2 and r['recency'] <= 90: return "Fidèle active"
    return "Fidèle en sommeil"
for r in rows: r['seg'] = seg(r)

seg_agg = defaultdict(lambda: {'n':0,'ca':0.0})
for r in rows:
    seg_agg[r['seg']]['n'] += 1
    seg_agg[r['seg']]['ca'] += r['monetary']
total_ca = sum(r['monetary'] for r in rows)
total_n = len(rows)

monetary_sorted = sorted(rows, key=lambda r:-r['monetary'])
top10n = max(1, round(total_n*0.10))
vip_ca_share = sum(r['monetary'] for r in monetary_sorted[:top10n]) / total_ca

print("=== Segments RFM ===")
for s,v in sorted(seg_agg.items(), key=lambda kv:-kv[1]['n']):
    print(f"  {s}: n={v['n']} ({v['n']*100/total_n:.1f}%) CA={v['ca']:.0f} ({v['ca']*100/total_ca:.1f}%) panier_moy_total={v['ca']/v['n']:.1f}")
print(f"VIP (10% meilleurs) part du CA: {vip_ca_share*100:.1f}%")

# ---- Cohortes d'acquisition : % avec >=2 commandes selon l'ancienneté disponible
cohort = defaultdict(list)
for r in rows:
    m = r['first'].strftime('%Y-%m')
    cohort[m].append(r)
print("\n=== Cohortes (mois de 1re commande) ===")
cohort_rows = []
for m, rs in sorted(cohort.items()):
    n = len(rs)
    rep = sum(1 for r in rs if r['freq']>=2)
    months_elapsed = (REF.year-int(m[:4]))*12 + (REF.month-int(m[5:7]))
    cohort_rows.append({'mois':m,'n':n,'reachat_pct':round(rep*100.0/n,1),'mois_recul':months_elapsed})
    print(f"  {m}: n={n} reachat={rep*100.0/n:.1f}% (recul {months_elapsed} mois)")

# ---- Marché de paniers : produits achetés ensemble (hors Tip)
pair_count = defaultdict(int)
item_count = defaultdict(int)
n_orders_with_2plus = 0
for o in orders:
    names = sorted(set(i['Lineitem name'] for i in o['items'] if i['Lineitem name'] and i['Lineitem name']!='Tip'))
    for n in names: item_count[n]+=1
    if len(names) >= 2:
        n_orders_with_2plus += 1
        for i in range(len(names)):
            for j in range(i+1, len(names)):
                pair_count[(names[i],names[j])] += 1
N = len(orders)
pairs_lift = []
for (a,b), c in pair_count.items():
    if c < 4: continue
    pa, pb = item_count[a]/N, item_count[b]/N
    lift = (c/N) / (pa*pb)
    pairs_lift.append({'a':a,'b':b,'n':c,'lift':round(lift,2)})
pairs_lift.sort(key=lambda p:-p['lift'])
print(f"\n=== Produits achetés ensemble (n commandes avec 2+ produits: {n_orders_with_2plus}) ===")
for p in pairs_lift[:12]:
    print(f"  {p['a'][:35]:35s} + {p['b'][:35]:35s} n={p['n']} lift={p['lift']}")

json.dump({
 'ref_date': REF.strftime('%Y-%m-%d'),
 'rfm_seg': [{'seg':s,'n':v['n'],'ca':round(v['ca'],2)} for s,v in sorted(seg_agg.items(), key=lambda kv:-kv[1]['n'])],
 'rfm_total_n': total_n, 'rfm_total_ca': round(total_ca,2), 'vip_ca_share': round(vip_ca_share*100,1),
 'cohort': cohort_rows,
 'pairs': pairs_lift[:15],
 'n_orders_multi': n_orders_with_2plus, 'n_orders_total': N,
}, open('behavior.json','w'), ensure_ascii=False)
