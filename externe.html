Données externes et géographie

Ce que des sources extérieures à la boutique apportent pour comprendre ses données, avec la source, sa fiabilité et ses limites.

Ce que ça mesure
Où habitent les acheteurs et à quel niveau de vie moyen, si la saisonnalité de la boutique suit celle du marché, et d’où viennent les visites.
Sur quel périmètre
Codes postaux de livraison en France (les clients belges, suisses et de la Réunion ne figurent pas sur la carte) ; recherches Google en France sur 15 mois.
Ce que ça permet de conclure
Voir où se concentre la clientèle, que le niveau de vie du territoire ne change pas le panier de façon nette, et que les pics de la boutique suivent la demande nationale de recherche.
Ce que ça ne permet pas de conclure
Décrire un client individuel (un code postal n’est pas une personne), ni prouver une cause : le croisement montre une coïncidence, pas un mécanisme.

import csv, json, math
from collections import defaultdict

# ---- 1) Table code postal -> (lat, lon) moyenne, et ville la plus fréquente
zips = defaultdict(list)
zip_city = defaultdict(lambda: defaultdict(int))
with open('geonames_fr/FR.txt', encoding='utf-8') as f:
    for row in csv.reader(f, delimiter='\t'):
        cp = row[1].strip().zfill(5)
        city = row[2]
        lat, lon = float(row[9]), float(row[10])
        zips[cp].append((lat, lon))
        zip_city[cp][city] += 1
zip_latlon = {cp: (sum(a for a,_ in v)/len(v), sum(b for _,b in v)/len(v)) for cp,v in zips.items()}
zip_bestcity = {cp: max(c.items(), key=lambda kv: kv[1])[0] for cp,c in zip_city.items()}
print("codes postaux connus:", len(zip_latlon))

# ---- 2) Commandes Lumineva : agréger par code postal (France, hors annulées)
import re
orders = []
cur = None
with open('../DONNEES/orders.csv', encoding='utf-8') as f:
    r = csv.DictReader(f)
    for row in r:
        if row['Total']:
            cur = row
            orders.append(row)
orders = [o for o in orders if not o.get('Cancelled at')]
print("commandes non annulées:", len(orders))

agg = defaultdict(lambda: {'n':0,'ca':0.0})
country_count = defaultdict(int)
unmatched = defaultdict(int)
for o in orders:
    country = (o.get('Shipping Country') or o.get('Billing Country') or '').strip()
    country_count[country or '(vide)'] += 1
    if country != 'FR':
        continue
    zip_ = (o.get('Shipping Zip') or o.get('Billing Zip') or '').strip()
    zip_ = re.sub(r'\D','',zip_).zfill(5)[:5]
    if len(zip_) != 5 or zip_ not in zip_latlon:
        unmatched[zip_] += 1
        continue
    agg[zip_]['n'] += 1
    agg[zip_]['ca'] += float(o['Total'])

print("pays:", dict(country_count))
matched_orders = sum(v['n'] for v in agg.values())
fr_orders = country_count.get('FR',0)
print(f"commandes FR geocodees: {matched_orders}/{fr_orders} ({matched_orders*100.0/fr_orders:.1f}%)")
print("codes postaux FR non trouves (top 10):", sorted(unmatched.items(), key=lambda x:-x[1])[:10])

points = []
for cp, v in agg.items():
    lat, lon = zip_latlon[cp]
    points.append({
        'cp': cp, 'ville': zip_bestcity.get(cp,''),
        'lat': lat, 'lon': lon,
        'n': v['n'], 'panier': round(v['ca']/v['n'], 2), 'ca': round(v['ca'],2)
    })
points.sort(key=lambda p:-p['n'])
print("points (codes postaux distincts):", len(points))
print("top 5:", points[:5])

json.dump({'points': points, 'matched_orders': matched_orders, 'fr_orders': fr_orders,
           'countries': country_count, 'unmatched_zip_count': sum(unmatched.values())},
          open('map_points.json','w'), ensure_ascii=False)

import json, math

DOM = {"Guadeloupe","Martinique","Guyane","La Réunion","Mayotte"}
gj = json.load(open('regions-avec-outre-mer.geojson'))
regions = [f for f in gj['features'] if f['properties'].get('nom') not in DOM]
print(len(regions), "régions métropolitaines")

pts = json.load(open('map_points.json'))['points']

# bbox sur les régions métropolitaines uniquement
lons, lats = [], []
def walk_rings(geom):
    if geom['type']=='Polygon':
        for ring in geom['coordinates']:
            yield ring
    elif geom['type']=='MultiPolygon':
        for poly in geom['coordinates']:
            for ring in poly:
                yield ring

for f in regions:
    for ring in walk_rings(f['geometry']):
        for lon,lat in ring:
            lons.append(lon); lats.append(lat)
lon_min,lon_max = min(lons),max(lons)
lat_min,lat_max = min(lats),max(lats)
lat0 = (lat_min+lat_max)/2
print("bbox lon", lon_min, lon_max, "lat", lat_min, lat_max)

W, H, PAD = 620, 620, 10
kx = math.cos(math.radians(lat0))
# échelle commune en x et y pour ne pas déformer
sx = (W-2*PAD)/((lon_max-lon_min)*kx)
sy = (H-2*PAD)/(lat_max-lat_min)
s = min(sx, sy)
def proj(lon, lat):
    x = PAD + (lon-lon_min)*kx*s
    y = H-PAD - (lat-lat_min)*s
    return x,y

def simplify(ring, min_d=2.0):
    out = []
    last = None
    pts_xy = [proj(lo,la) for lo,la in ring]
    for x,y in pts_xy:
        if last is None or (x-last[0])**2+(y-last[1])**2 >= min_d*min_d:
            out.append((x,y)); last=(x,y)
    if len(out) < 3:
        out = pts_xy[::max(1,len(pts_xy)//4)]
    return out

def path_d(feature):
    d = []
    for ring in walk_rings(feature['geometry']):
        pts_s = simplify(ring)
        if len(pts_s) < 3: continue
        d.append("M"+" L".join(f"{x:.1f},{y:.1f}" for x,y in pts_s)+" Z")
    return " ".join(d)

region_paths = []
total_pts = 0
for f in regions:
    dstr = path_d(f)
    total_pts += dstr.count("L")+dstr.count("M")
    region_paths.append({"nom": f['properties'].get('nom'), "d": dstr})
print("total points après simplification:", total_pts)
print("taille totale d (Ko):", sum(len(r['d']) for r in region_paths)/1024)

# points clients projetés
proj_pts = []
for p in pts:
    x,y = proj(p['lon'], p['lat'])
    if 0 <= x <= W and 0 <= y <= H:
        proj_pts.append({**p, 'x': round(x,1), 'y': round(y,1)})
    else:
        proj_pts.append({**p, 'x': round(x,1), 'y': round(y,1), 'out': True})
out_of_bounds = sum(1 for p in proj_pts if p.get('out'))
print("points hors cadre (DOM/erreur):", out_of_bounds)

json.dump({'W':W,'H':H,'regions':region_paths,'points':proj_pts}, open('map_svg_data.json','w'), ensure_ascii=False)

import csv, json, time
from datetime import datetime
from collections import defaultdict
from pytrends.request import TrendReq
from scipy import stats

# ---- commandes Lumineva par mois
orders = []
with open('../DONNEES/orders.csv', encoding='utf-8') as f:
    for row in csv.DictReader(f):
        if row['Total'] and not row.get('Cancelled at'):
            orders.append(row)
mo = defaultdict(int)
for o in orders:
    mo[o['Created at'][:7]] += 1
months = sorted(mo.keys())
print("mois lumineva:", months)

# ---- Google Trends mensuel, France, sur toute la période
pt = TrendReq(hl='fr-FR', tz=60)
kw = ['bougie parfumée', 'fondant parfumé']
pt.build_payload(kw, timeframe='2025-06-01 2026-09-21', geo='FR')
df = pt.interest_over_time()
print(df.head(3)); print(df.tail(3))

trend_m = defaultdict(lambda: defaultdict(list))
for idx, row in df.iterrows():
    m = idx.strftime('%Y-%m')
    for k in kw:
        trend_m[k][m].append(row[k])
trend_month = {k: {m: sum(v)/len(v) for m,v in d.items()} for k,d in trend_m.items()}

common = [m for m in months if m in trend_month['bougie parfumée']]
print("mois communs:", len(common))
ord_vals = [mo[m] for m in common]
res = {}
for k in kw:
    tv = [trend_month[k][m] for m in common]
    r, p = stats.pearsonr(ord_vals, tv)
    res[k] = {'r': round(r,3), 'p': round(p,4)}
    print(f"corrélation commandes <-> recherches '{k}': r={r:.3f} p={p:.4f}")
    # avec décalage +1 mois (recherche precede la commande ?)
    if len(common) > 2:
        r1,p1 = stats.pearsonr(ord_vals[1:], tv[:-1])
        res[k]['r_lag1'] = round(r1,3); res[k]['p_lag1'] = round(p1,4)
        print(f"   décalage recherche(m-1) -> commandes(m): r={r1:.3f} p={p1:.4f}")

json.dump({'months': common, 'orders': ord_vals,
           'trend': {k: [round(trend_month[k][m],1) for m in common] for k in kw},
           'corr': res}, open('a4_trends.json','w'), ensure_ascii=False)

"""Analyse du rapport Shopify 'Visites par référent' fourni par l'enseignant le 22/09.
ATTENTION : ce fichier ne contient aucune date -> la période exacte est inconnue,
à confirmer avec l'équipe avant toute conclusion liée à une date précise."""
import csv
from collections import defaultdict

SRC = "../DONNEES/visites_par_referent.csv"
rows = list(csv.DictReader(open(SRC, encoding='utf-8')))

def to_i(s):
    try: return int(s)
    except: return 0

tot_visits = sum(to_i(r['Visites']) for r in rows)
tot_prev = sum(to_i(r['Sessions (previous_period)']) for r in rows)
print(f"Total visites (période actuelle): {tot_visits} | (période précédente): {tot_prev} | évolution: {(tot_visits/tot_prev-1)*100:+.1f}%")

by_source = defaultdict(lambda: {'visits': 0, 'prev': 0})
for r in rows:
    s = r['Source du référent']
    by_source[s]['visits'] += to_i(r['Visites'])
    by_source[s]['prev'] += to_i(r['Sessions (previous_period)'])
for s, v in sorted(by_source.items(), key=lambda kv: -kv[1]['visits']):
    chg = f"{(v['visits']/v['prev']-1)*100:+.0f}%" if v['prev'] else 'n/a'
    print(f"  {s:10s} visites={v['visits']:4d} (précédente={v['prev']:4d}, {chg})")

# indices de trafic non-humain (villes = data centers connus) parmi le trafic 'direct'
DATACENTER_CITIES = {'Boardman', 'Council Bluffs', 'Ashburn', 'The Dalles', 'Moncks Corner', 'North Charleston'}
suspect = [r for r in rows if r['Source du référent'] == 'direct' and r['Ville de la visite'] in DATACENTER_CITIES]
susp_visits = sum(to_i(r['Visites']) for r in suspect)
print(f"\nVisites 'direct' associées à des villes de data centers connus (Boardman OR, Council Bluffs IA...): {susp_visits} sur {by_source['direct']['visits']} direct au total")
print("Lignes concernées:", [(r['Ville de la visite'], r['Visites']) for r in suspect])
