import csv, json
from datetime import datetime, date
from collections import defaultdict

def parse_dt(s):
    # "2026-09-20 20:42:31 +0200"
    return datetime.strptime(s[:19], "%Y-%m-%d %H:%M:%S")

orders = []
with open('../DONNEES/orders.csv', encoding='utf-8') as f:
    r = csv.DictReader(f)
    cur = None
    for row in r:
        if row['Total']:
            cur = {'r': row, 'items': []}
            orders.append(cur)
        if cur: cur['items'].append(row)
orders = [o for o in orders if not o['r'].get('Cancelled at')]
for o in orders:
    o['dt'] = parse_dt(o['r']['Created at'])
    o['tot'] = float(o['r']['Total'])
    o['key'] = o['r']['customer_key'].strip()

REF = datetime(2026,9,21)  # lendemain de la dernière commande (20/09/2026)

cust = defaultdict(list)
for o in orders:
    if o['key']:
        cust[o['key']].append(o)
for v in cust.values():
    v.sort(key=lambda o:o['dt'])

# ---- RFM pragmatique (4 segments + VIP transversal)
rows = []
for k, os_ in cust.items():
    freq = len(os_)
    monetary = sum(o['tot'] for o in os_)
    recency = (REF - os_[-1]['dt']).days
    first = os_[0]['dt']
    rows.append({'key':k,'freq':freq,'monetary':monetary,'recency':recency,'first':first})

def seg(r):
    if r['freq'] == 1 and r['recency'] > 180: return "Perdue après 1 achat"
    if r['freq'] == 1: return "Nouvelle (1 achat récent)"
    if r['freq'] >= 2 and r['recency'] <= 90: return "Fidèle active"
    return "Fidèle en sommeil"
for r in rows: r['seg'] = seg(r)

seg_agg = defaultdict(lambda: {'n':0,'ca':0.0})
for r in rows:
    seg_agg[r['seg']]['n'] += 1
    seg_agg[r['seg']]['ca'] += r['monetary']
total_ca = sum(r['monetary'] for r in rows)
total_n = len(rows)

monetary_sorted = sorted(rows, key=lambda r:-r['monetary'])
top10n = max(1, round(total_n*0.10))
vip_ca_share = sum(r['monetary'] for r in monetary_sorted[:top10n]) / total_ca

print("=== Segments RFM ===")
for s,v in sorted(seg_agg.items(), key=lambda kv:-kv[1]['n']):
    print(f"  {s}: n={v['n']} ({v['n']*100/total_n:.1f}%) CA={v['ca']:.0f} ({v['ca']*100/total_ca:.1f}%) panier_moy_total={v['ca']/v['n']:.1f}")
print(f"VIP (10% meilleurs) part du CA: {vip_ca_share*100:.1f}%")

# ---- Cohortes d'acquisition : % avec >=2 commandes selon l'ancienneté disponible
cohort = defaultdict(list)
for r in rows:
    m = r['first'].strftime('%Y-%m')
    cohort[m].append(r)
print("\n=== Cohortes (mois de 1re commande) ===")
cohort_rows = []
for m, rs in sorted(cohort.items()):
    n = len(rs)
    rep = sum(1 for r in rs if r['freq']>=2)
    months_elapsed = (REF.year-int(m[:4]))*12 + (REF.month-int(m[5:7]))
    cohort_rows.append({'mois':m,'n':n,'reachat_pct':round(rep*100.0/n,1),'mois_recul':months_elapsed})
    print(f"  {m}: n={n} reachat={rep*100.0/n:.1f}% (recul {months_elapsed} mois)")

# ---- Marché de paniers : produits achetés ensemble (hors Tip)
pair_count = defaultdict(int)
item_count = defaultdict(int)
n_orders_with_2plus = 0
for o in orders:
    names = sorted(set(i['Lineitem name'] for i in o['items'] if i['Lineitem name'] and i['Lineitem name']!='Tip'))
    for n in names: item_count[n]+=1
    if len(names) >= 2:
        n_orders_with_2plus += 1
        for i in range(len(names)):
            for j in range(i+1, len(names)):
                pair_count[(names[i],names[j])] += 1
N = len(orders)
pairs_lift = []
for (a,b), c in pair_count.items():
    if c < 4: continue
    pa, pb = item_count[a]/N, item_count[b]/N
    lift = (c/N) / (pa*pb)
    pairs_lift.append({'a':a,'b':b,'n':c,'lift':round(lift,2)})
pairs_lift.sort(key=lambda p:-p['lift'])
print(f"\n=== Produits achetés ensemble (n commandes avec 2+ produits: {n_orders_with_2plus}) ===")
for p in pairs_lift[:12]:
    print(f"  {p['a'][:35]:35s} + {p['b'][:35]:35s} n={p['n']} lift={p['lift']}")

json.dump({
 'ref_date': REF.strftime('%Y-%m-%d'),
 'rfm_seg': [{'seg':s,'n':v['n'],'ca':round(v['ca'],2)} for s,v in sorted(seg_agg.items(), key=lambda kv:-kv[1]['n'])],
 'rfm_total_n': total_n, 'rfm_total_ca': round(total_ca,2), 'vip_ca_share': round(vip_ca_share*100,1),
 'cohort': cohort_rows,
 'pairs': pairs_lift[:15],
 'n_orders_multi': n_orders_with_2plus, 'n_orders_total': N,
}, open('behavior.json','w'), ensure_ascii=False)

import csv, json
from datetime import datetime
from collections import defaultdict
from scipy import stats
from svgutil import *

orders = []
with open('../DONNEES/orders.csv', encoding='utf-8') as f:
    r = csv.DictReader(f)
    cur = None
    for row in r:
        if row['Total']:
            cur = {'r': row, 'items': []}
            orders.append(cur)
        if cur: cur['items'].append(row)
orders = [o for o in orders if not o['r'].get('Cancelled at')]
for o in orders:
    o['dt'] = datetime.strptime(o['r']['Created at'][:19], "%Y-%m-%d %H:%M:%S")
    o['tot'] = float(o['r']['Total'])
    o['key'] = o['r']['customer_key'].strip()
    o['disc_amt'] = float(o['r']['Discount Amount'] or 0)
    o['code'] = o['r']['Discount Code']

cust = defaultdict(list)
for o in orders:
    if o['key']: cust[o['key']].append(o)
for v in cust.values(): v.sort(key=lambda o:o['dt'])

groupA, groupB = [], []  # A = 1re commande avec BIENVENUE10, B = 1re commande sans code
disc_cost_A = 0.0
for k, os_ in cust.items():
    first = os_[0]
    lifetime = sum(o['tot'] for o in os_)
    if first['code'] == 'BIENVENUE10':
        groupA.append(lifetime)
        disc_cost_A += first['disc_amt']
    elif not first['code']:
        groupB.append(lifetime)

t_stat, p_val = stats.ttest_ind(groupA, groupB, equal_var=False)  # Welch
mean_a, mean_b = sum(groupA)/len(groupA), sum(groupB)/len(groupB)
med = lambda a: sorted(a)[len(a)//2] if len(a)%2 else (sorted(a)[len(a)//2-1]+sorted(a)[len(a)//2])/2
avg_disc_per_customer = disc_cost_A/len(groupA)
clv_net_A = mean_a - avg_disc_per_customer

print(f"Groupe A (1re cmd BIENVENUE10): n={len(groupA)} CLV moyenne={mean_a:.2f} médiane={med(groupA):.2f}")
print(f"Groupe B (1re cmd sans code):   n={len(groupB)} CLV moyenne={mean_b:.2f} médiane={med(groupB):.2f}")
print(f"Welch t-test: t={t_stat:.3f} p={p_val:.4f}")
print(f"Coût total des remises BIENVENUE10 attribuables à ces {len(groupA)} clients: {disc_cost_A:.2f} €")
print(f"Coût moyen par client acquis via le code: {avg_disc_per_customer:.2f} €")
print(f"CLV nette (CLV - coût d'acquisition) groupe A: {clv_net_A:.2f} € vs groupe B (coût 0): {mean_b:.2f} €")

json.dump({
 'n_a': len(groupA), 'n_b': len(groupB),
 'mean_a': round(mean_a,2), 'mean_b': round(mean_b,2),
 'med_a': round(med(groupA),2), 'med_b': round(med(groupB),2),
 't_stat': round(t_stat,3), 'p_value': round(p_val,4),
 'disc_cost_total': round(disc_cost_A,2), 'disc_cost_per_customer': round(avg_disc_per_customer,2),
 'clv_net_a': round(clv_net_A,2),
}, open('a1_clv.json','w'), ensure_ascii=False)

from datetime import datetime
from collections import defaultdict
from scipy import stats

with open('../DONNEES/orders.csv', encoding='utf-8') as f:
    r = csv.DictReader(f)
    cur = None
    for row in r:
        if row['Total']:
            cur = {'r': row, 'items': []}
            orders.append(cur)
        if cur: cur['items'].append(row)
orders = [o for o in orders if not o['r'].get('Cancelled at')]
for o in orders:
    o['dt'] = datetime.strptime(o['r']['Created at'][:19], "%Y-%m-%d %H:%M:%S")
    o['key'] = o['r']['customer_key'].strip()
    o['code'] = o['r']['Discount Code']

cust = defaultdict(list)
for o in orders:
    if o['key']: cust[o['key']].append(o)
for v in cust.values(): v.sort(key=lambda o:o['dt'])

REF = datetime(2026,9,21)

def build_events(customers):
    """Retourne [(duree_jours, evenement)] : evenement=1 si 2e commande observée, 0 si censuré."""
    ev = []
    for os_ in customers:
        first = os_[0]['dt']
        if len(os_) >= 2:
            ev.append(((os_[1]['dt']-first).days, 1))
        else:
            ev.append(((REF-first).days, 0))
    return ev

def kaplan_meier(events, tmax=270):
    """Estimateur de Kaplan-Meier : renvoie liste de (t, %ayant_recommandé) en escalier."""
    times = sorted(set(t for t,e in events if e==1 and t<=tmax))
    n_at_risk = len(events)
    s = 1.0
    curve = [(0,0.0)]
    remaining = sorted(events, key=lambda x:x[0])
    idx = 0
    at_risk = len(events)
    for t in times:
        d = sum(1 for tt,e in events if tt==t and e==1)
        # nombre encore à risque juste avant t : tous ceux dont la durée >= t
        at_risk_t = sum(1 for tt,e in events if tt >= t)
        if at_risk_t > 0:
            s *= (1 - d/at_risk_t)
        curve.append((t, round((1-s)*100,2)))
    curve.append((tmax, curve[-1][1]))
    return curve

def logrank(evA, evB):
    """Test du log-rank (comparaison de 2 courbes de survie), khi² à 1 ddl."""
    all_times = sorted(set(t for t,e in evA+evB if e==1))
    O_A=E_A=V=0.0
    for t in all_times:
        nA = sum(1 for tt,e in evA if tt>=t); nB = sum(1 for tt,e in evB if tt>=t)
        dA = sum(1 for tt,e in evA if tt==t and e==1); dB = sum(1 for tt,e in evB if tt==t and e==1)
        n, d = nA+nB, dA+dB
        if n <= 1 or d == 0: continue
        O_A += dA
        E_A += d*nA/n
        V += d*(nA/n)*(nB/n)*((n-d)/(n-1))
    chi2 = (O_A-E_A)**2/V if V>0 else 0
    p = stats.chi2.sf(chi2, df=1)
    return chi2, p

evA = build_events([os_ for os_ in cust.values() if os_[0]['code']=='BIENVENUE10'])
evB = build_events([os_ for os_ in cust.values() if not os_[0]['code']])
evAll = build_events(list(cust.values()))

curveA = kaplan_meier(evA)
curveB = kaplan_meier(evB)
curveAll = kaplan_meier(evAll)
chi2, p = logrank(evA, evB)

def median_time(curve):
    for t,y in curve:
        if y >= 50: return t
    return None

print(f"Tous clients (n={len(evAll)}): réachat à 30j={dict(curveAll)[min([t for t,_ in curveAll if t>=30], default=curveAll[-1][0])]}%")
print(f"Groupe BIENVENUE10 (n={len(evA)}): médiane (50%) atteinte à t={median_time(curveA)} j" if median_time(curveA) else f"BIENVENUE10: jamais 50%, plateau à {curveA[-1][1]}%")
print(f"Groupe sans code (n={len(evB)}): médiane (50%) atteinte à t={median_time(curveB)} j" if median_time(curveB) else f"sans code: jamais 50%, plateau à {curveB[-1][1]}%")
print(f"Log-rank test A vs B: chi2={chi2:.3f} p={p:.4f}")

def pct_at(curve, days):
    val = 0
    for t,y in curve:
        if t<=days: val=y
    return val
for d in (30,60,90,180,270):
    print(f"  à {d}j: tous={pct_at(curveAll,d)}%  BIENVENUE10={pct_at(curveA,d)}%  sans_code={pct_at(curveB,d)}%")

json.dump({'curveA': curveA, 'curveB': curveB, 'curveAll': curveAll,
           'n_a': len(evA), 'n_b': len(evB), 'n_all': len(evAll),
           'chi2': round(chi2,3), 'p_logrank': round(p,4),
           'pct_at': {str(d): {'all':pct_at(curveAll,d),'a':pct_at(curveA,d),'b':pct_at(curveB,d)} for d in (30,60,90,180,270)}},
          open('a3_survival.json','w'), ensure_ascii=False)
