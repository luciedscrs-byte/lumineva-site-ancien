def lecture(t): return f'<p class="hint" style="margin-top:10px">{t}</p>'
def cols(a, b): return f'<div style="display:grid;grid-template-columns:repeat(auto-fit,minmax(280px,1fr));gap:16px;margin-top:10px">{a}{b}</div>'
def box(titre, items):

def arbre_html(nd, label=''):
    tete = f'<span class="tnode"><b>{nd["taux"]:.0f} %</b> reviennent <span class="hint">({nd["n"]} clients)</span></span>'
    if 'var' not in nd:
        return f'<li>{label}{tete}</li>'
    if nd['bin']: q_non, q_oui = 'non', 'oui'
    else:
        sv = nd['seuil']; sv = f'{int(sv)}' if abs(sv - int(sv) - 0.5) < 1e-6 else n(sv, 1)
        q_non, q_oui = f'≤ {sv}', f'&gt; {sv}'
    q = f'<span class="tq">{esc(nd["var"])}{" ?" if nd["bin"] else ""}</span>'
    return f'<li>{label}{tete}<div class="tq2">{q}</div><ul>{arbre_html(nd["non"], f"<i class=tl>{q_non}</i> ")}{arbre_html(nd["oui"], f"<i class=tl>{q_oui}</i> ")}</ul></li>'

        for f in nd['fam']:
