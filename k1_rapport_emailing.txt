def esc(t): return t.replace('&', '&amp;').replace('<', '&lt;').replace('>', '&gt;')
def jj(date): return datetime.strptime(date[:10], '%Y-%m-%d').strftime('%d/%m/%y')
def court(o, k=44): return o if len(o) <= k else o[:k - 1].rstrip() + '…'
def lab(e): return jj(e['date']) + ' · ' + court(re.sub(r'\s+', ' ', e['objet']), 38)


# ------------------------------------------------------------------ graphiques
def graphe_clic(lst, ref):
    rows = []
    for e in lst:
        hors = e['clic_lo'] > ref or e['clic_hi'] < ref
        rows.append({'label': lab(e), 'v': e['clic'], 'lo': e['clic_lo'], 'hi': e['clic_hi'], 'p': 0.01 if hors else 1.0})
    xmax = max(r['hi'] for r in rows) + 1; pas = 2 if xmax <= 10 else 5
    ticks = [t for t in range(0, 31, pas) if t <= xmax]
    return forest(rows, ref, 0, xmax, ticks, lambda v: n(v, 0).replace(' ', ' ') + ' %', w=900, label_w=330, titre_axe='taux de clic, intervalle à 95 % ; trait noir = moyenne de la liste')

def graphe_ca(lst, ref):
    rows = []
    for e in lst:
        hors = e['ca_lo'] > ref or e['ca_hi'] < ref
        rows.append({'label': lab(e), 'v': e['ca_pm'], 'lo': e['ca_lo'], 'hi': min(e['ca_hi'], 700), 'p': 0.01 if hors else 1.0})
    return forest(rows, ref, 0, 700, [0, 100, 200, 300, 400, 500, 600, 700], lambda v: n(v, 0).replace(' ', ' ') + ' €', w=900, label_w=330, titre_axe='chiffre d’affaires attribué pour 1 000 e-mails envoyés (intervalle à 95 %, tronqué à 700 €)')

def graphe_auto():
    rows = []
    for e in AU:
        if e['envoyes'] > 100:
            rows.append({'label': e['automatisation'], 'sous': n(e['envoyes']) + ' envois · ' + n(e['cmd']) + ' commandes', 'v': e['ca_pm'], 'lo': e['ca_lo'], 'hi': min(e['ca_hi'], 3600), 'p': 0.01 if e['ca_lo'] > G['boutique']['ca_pm'] else 1.0})
    rows.append({'label': 'Newsletters boutique (12 envois)', 'sous': n(G['boutique']['envoyes']) + ' envois · ' + n(G['boutique']['cmd']) + ' commandes', 'v': G['boutique']['ca_pm'], 'lo': G['boutique']['ca_pm_lo'], 'hi': G['boutique']['ca_pm_hi'], 'p': 1.0})
    return forest(rows, 0, 0, 3600, [0, 500, 1000, 1500, 2000, 2500, 3000, 3500], lambda v: n(v, 0).replace(' ', ' ') + ' €', w=900, label_w=330, titre_axe='chiffre d’affaires attribué pour 1 000 e-mails envoyés (intervalle à 95 %)')

def graphe_fatigue():
    auto = [e for e in BOUT if e['date'] >= '2026-09-10']
    vals = [e['clic'] for e in auto]; labs = [jj(e['date']) for e in auto]
    return svgutil.vbars(vals, labs, 640, 250, lambda v: n(v, 0) + ' %', 99, 2.0, rot=True, colors=['--flame' for _ in vals])

def graphe_evenement():
    rows = [{'label': 'le jour de l’envoi', 'v': ev['jour d’envoi']['rr'], 'lo': ev['jour d’envoi']['lo'], 'hi': ev['jour d’envoi']['hi'], 'p': ev['jour d’envoi']['p']},

    return forest(rows, 1, 0.6, 4.0, [0.7, 1, 2, 3, 4], lambda v: '× ' + n(v, 1).replace(' ', ' '), w=900, label_w=250, titre_axe='multiplicateur des commandes de toute la boutique')


# ------------------------------------------------------------------ tableaux
def table(head, rows, right=()):
    h = ''.join('<th class="%s">%s</th>' % ('r' if i in right else '', c) for i, c in enumerate(head))
    b = ''.join('<tr>' + ''.join('<td class="%s">%s</td>' % ('r' if i in right else '', c) for i, c in enumerate(r)) + '</tr>' for r in rows)
    return '<div class="tw"><table><tr>%s</tr>%s</table></div>' % (h, b)

def tab_news(lst):
    rows = []
    for e in lst:
        rows.append([jj(e['date']), esc(court(re.sub(r'\s+', ' ', e['objet']), 52)), n(e['envoyes']), pct(e['ouv'], 1), pct(e['clic'], 2), n(e['cmd']), eur(e['ca']), eur(e['ca_pm']) if e['envoyes'] else '—',
                     esc(e['theme']), n(e['nb_images']), esc(e['gras'].replace('nombreux passages', 'beaucoup').replace('quelques mots', 'un peu')), n(e['mots'])])
    return table(['Date', 'Objet', 'Envois', 'Ouverture', 'Clic', 'Cmd', 'CA', 'CA / 1 000 envois', 'Thème', 'Photos', 'Gras', 'Mots'], rows, right=(2, 3, 4, 5, 6, 7, 9, 11))

def tab_criteres():
    rows = []
    for c in d['criteres']:
        signif = c['h_clic'] < 0.05
        conf = abs(c['rho_temps']) >= 0.5
        lect = ('<b>Écart net</b>' if signif else 'Écart non démontrable') + ('<br><span class="petit">se confond avec la date (' + signe(c['rho_temps'], 2) + ')</span>' if conf else '')
        rows.append([esc(c['nom']), '%d / %d' % (c['n_oui'], c['n_non']), pct(c['clic_oui'], 2) + ' contre ' + pct(c['clic_non'], 2), pv(c['p_clic']) + ' <span class="petit">(corrigée ' + pv(c['h_clic']) + ')</span>',
                     n(c['cmd_pm_oui'], 2) + ' contre ' + n(c['cmd_pm_non'], 2), n(c['p_cmd'], 2), lect])
    return table(['Caractéristique de l’e-mail', 'Envois avec / sans', 'Clic avec contre sans', 'p-value du clic', 'Commandes pour 1 000 envois', 'p (commandes)', 'Lecture'], rows, right=(1,))

def vignette(e, rang):

            % (e['id'], esc(e['image']), rang, esc(court(re.sub(r'\s+', ' ', e['objet']), 46)), pct(e['clic'], 2), eur(e['ca']) + ' attribués'))
