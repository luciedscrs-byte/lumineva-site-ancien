Avant toute analyse : quelles données, quelles règles de comptage, quels contrôles. Un périmètre faux fausse tout le reste, c’est donc la page la plus importante du site.

Règles de comptage
  • Une commande = une clé de commande. Les lignes d’un même achat sont regroupées ; les 7 commandes annulées sont retirées, ce qui laisse 1 438 commandes.
  • La ligne « pourboire » n’est pas un produit : elle est exclue des contenus de commande.
  • 25 commandes n’ont pas d’identifiant client (dont des ventes en personne) : elles comptent dans les ventes, pas dans le suivi des clients (829 clients suivis).
  • La TVA n’est facturée que depuis le 28 mai 2026. Le chiffre d’affaires facturé (90 721 €) mélange donc des montants sans TVA puis avec TVA ; hors TVA il est de 87 036 €.
  • Le recul compte. Un client arrivé il y a trois semaines n’a pas eu le temps de recommander : les taux de retour ne sont comparés qu’avec au moins 90 jours de recul.
  • Seuls des agrégats sont affichés : aucun nom, e-mail, téléphone ni adresse.

Ce que le périmètre ne couvre pas
  • Marge : seule la marge de 50 % sur les bougies (indiquée par la gérante) est connue ; la colonne « coût par article » du catalogue est vide. Les bougies font 71 % du chiffre d’affaires des lignes de commande.
  • Coût des colis : les fichiers donnent le port facturé aux clients (5,90 € en point relais, 9,90 € à domicile ; 5 648 € au total), pas ce que la boutique paie au transporteur.
  • Aucune donnée de trafic datée, de conversion ni de provenance par commande : la colonne « Source » des commandes ne contient aucune valeur Instagram.
  • Aucune information sur les actions marketing (e-mails, publicités, calendrier de lancement).
  • Quinze mois d’historique : les corrélations calculées portent sur très peu de points.

# encoding: utf-8
"""Base commune aux analyses de parcours client : lecture de orders.csv (hors commandes annulées),
familles de produits, clés produit, commandes rangées par client."""
import csv, re
from datetime import datetime
from collections import defaultdict

ORDERS_CSV = '../DONNEES/orders.csv'
REF = datetime(2026, 10, 23)

FAMILLES = ['Bougies gourmandes', 'Bougies boisson', 'Bougies taille S', 'Autres bougies', 'Fondants',
            'Voiture', 'Diffuseurs maison', 'Poudre parfumée', 'Élixirs', 'Accessoires et cadeaux', 'Formation']

def famille(name):
    n = name.upper()
    if n.startswith('BOUGIE GOURMANDE'): return 'Bougies gourmandes'
    if n.startswith(('BOUGIE BOISSON', 'BOUGIE LATTE', 'BOUGIE ICED')): return 'Bougies boisson'
    if n.startswith('BOUGIE PARFUMEE TAILLE S'): return 'Bougies taille S'
    if n.startswith('BOUGIE'): return 'Autres bougies'
    if n.startswith(('BOX DE FONDANTS', 'BOUQUET DE FONDANTS', 'TABLETTE', 'FONDANT')): return 'Fondants'
    if n.startswith(('SUSPENSION VOITURE', 'DIFFUSEUR VOITURE')): return 'Voiture'
    if n.startswith(('DIFFUSEUR', 'RECHARGE MELANGE')): return 'Diffuseurs maison'
    if n.startswith('POUDRE'): return 'Poudre parfumée'
    if n.startswith('ELIXIR'): return 'Élixirs'
    if n.startswith(('PACK FABRICATION', 'PACK BOUGIES')) or 'CONSULTING' in n or n.startswith('RÉGLEMENTATION') or n.startswith('REGLEMENTATION'): return 'Formation'
    return 'Accessoires et cadeaux'

def cle_produit(name):
    """Produit au sens marketing : on retire la variante après ' - ' ou ' / ' (parfum, taille)."""
    return re.split(r' - | / ', name)[0].strip()

def joli(s):
    t = s.replace('BOUGIE GOURMANDE ', 'Bougie ')
    t = t.translate(str.maketrans('ÉÈÊÀÂÎÔÛÇÏËÖ', 'éèêàâîôûçïëö')).lower()
    return t[0].upper() + t[1:]

def charger():
    """Retourne (commandes, clients) : commandes = liste de dicts ; clients = {cle: [commandes triées par date]}."""
    rows = []
    cur = None
    with open(ORDERS_CSV, encoding='utf-8') as f:
        for r in csv.DictReader(f):
            if r['Total']:
                cur = {'r': r, 'lignes': []}
                rows.append(cur)
            if cur and r['Lineitem name'] and r['Lineitem name'] != 'Tip':
                cur['lignes'].append({'nom': r['Lineitem name'], 'qte': int(r['Lineitem quantity']), 'prix': float(r['Lineitem price'])})
    commandes = []
    for o in rows:
        r = o['r']
        if r.get('Cancelled at'): continue
        sm = r['Shipping Method'] or ''
        d = {
            'date': r['Created at'][:10], 'dt': datetime.strptime(r['Created at'][:19], '%Y-%m-%d %H:%M:%S'),
            'cle': r['customer_key'].strip(), 'total': float(r['Total']), 'sous_total': float(r['Subtotal']),
            'code': r['Discount Code'] or '', 'remise': float(r['Discount Amount'] or 0),
            'local': 1 if re.match(r'^\d|otissement', sm) else 0,
            'pays': (r['Shipping Country'] or r['Billing Country'] or '').strip(),
            'cp': re.sub(r'\D', '', (r['Shipping Zip'] or r['Billing Zip'] or '')).zfill(5)[:5],
            'ordre_key': r['order_key'],
            'lignes': o['lignes'],
        }
        d['qte'] = sum(l['qte'] for l in d['lignes'])
        d['familles'] = sorted(set(famille(l['nom']) for l in d['lignes']))
        d['produits'] = sorted(set(cle_produit(l['nom']) for l in d['lignes']))
        val = defaultdict(float)
        for l in d['lignes']: val[famille(l['nom'])] += l['qte'] * l['prix']
        d['fam_dom'] = max(val.items(), key=lambda kv: kv[1])[0] if val else None
        d['prix_unit'] = (d['sous_total'] / d['qte']) if d['qte'] else None
        commandes.append(d)
    clients = defaultdict(list)
    for d in commandes:
        if d['cle'] and d['lignes']: clients[d['cle']].append(d)
    for v in clients.values(): v.sort(key=lambda d: d['dt'])
    for v in clients.values():
        for i, d in enumerate(v): d['rang'] = i + 1
    return commandes, clients

    o['country'] = (r['Shipping Country'] or r['Billing Country'] or '').strip()

    elif 'domicile' in sm.lower(): o['shipmode']='mondial_domicile'

    o['pay'] = pay.get(r['order_key'], None)

    o['is_weekend'] = 1 if o['weekday'] >= 5 else 0
    o['month'] = r['Created at'][:7]

# ordre chronologique par client -> repeat/new

for o in orders:

for v in cust.values(): v.sort(key=lambda o:o['dt'])
